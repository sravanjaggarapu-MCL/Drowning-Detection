pillow

# RTSP capture and JPEG encoding (real mode).
opencv-python

# YOLO person detection (real mode).
# Installs PyTorch as a dependency; on a Pi this can take a
# while. Use the "yolov8n" (nano) model for CPU inference.
ultralytics
